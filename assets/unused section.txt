  function filterCards(tag, clickedBtn) {
    const container = document.getElementById('filteredCards');
    container.innerHTML = ''; // clear previous filtered content

    // Reset all buttons
    const buttons = document.querySelectorAll('#filterButtons .contact-btn');
    buttons.forEach(btn => {
      btn.style.background = '';
      btn.style.color = '';
    });

    // Highlight clicked button
    clickedBtn.style.background = 'var(--white)';
    clickedBtn.style.color = 'var(--secondary)';

    // Add close button
    const closeBtn = document.createElement('button');
    closeBtn.textContent = 'Clear Content';
    closeBtn.style.alignSelf = 'flex-start';
    closeBtn.style.background = 'maroon';
    closeBtn.style.color = 'white';
    closeBtn.style.border = 'none';
    closeBtn.style.borderRadius = '4px';
    closeBtn.style.padding = '0.25rem 0.5rem';
    closeBtn.style.cursor = 'pointer';
    closeBtn.onclick = () => {
      container.innerHTML = '';
      // Reset all buttons
      buttons.forEach(btn => {
        btn.style.background = '';
        btn.style.color = '';
      });
    };
    container.appendChild(closeBtn);

    // Filter and add links
    const filtered = cards.filter(card => card.tags.includes(tag));
    filtered.forEach(card => {
      const div = document.createElement('div');
      div.style.display = 'flex';
      div.style.gap = '1rem';
      div.style.alignItems = 'center';

      const a = document.createElement('a');
      a.textContent = `View: ${card.title}`;
      a.style.color = "darkblue";
      a.style.textDecoration = "underline";
      a.style.cursor = "pointer";

      // <-- NEW: handle protected card
      if (card.protected) {
        a.href = "#";
        a.onclick = (e) => {
          e.preventDefault();
          openProtectedPage();
        };
      } else {
        a.href = card.link;
        a.target = "_blank";
      }

      div.appendChild(a);
      container.appendChild(div);
    });
  }
